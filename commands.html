Commands

Complete reference for all CLI options

Basic Commands

bash
# Run with debug window
python src/main.py --debug

# Run headless (no display)
python src/main.py

# Show version
python src/main.py --version

# Show system info
python src/main.py --info

Camera Options

OptionDescriptionDefault
--camera CAMERACamera index or RTSP URL0
--width WIDTHFrame width640
--height HEIGHTFrame height480
--fps FPSTarget FPS30

Model Options

bash
# Disable specific models
python src/main.py --no-face
python src/main.py --no-gesture
python src/main.py --no-pose
python src/main.py --no-depth

# Run pose every N frames
python src/main.py --pose-every 3

Performance Options

bash
# Disable parallel processing
python src/main.py --no-parallel

# Model precision
python src/main.py --precision fp32
python src/main.py --precision fp16
python src/main.py --precision int8

# Use DLA (Deep Learning Accelerator)
python src/main.py --dla

# Batch size
python src/main.py --batch-size 2

Tracking Options

bash
# Enable person following
python src/main.py --follow

# Disable tracking
python src/main.py --no-tracking

# Set max tracking age
python src/main.py --follow --track-max-age 60

Model Selection

bash
# List available models
python src/main.py --list-models

# Select YOLO model
python src/main.py --model yolo12n
python src/main.py --model yolov8n
python src/main.py --model rtmdet_nano

Advanced AI

bash
# Enable VLA processing
python src/main.py --vla

# Enable all AI features
python src/main.py --advanced-ai

# Use real VLA models
python src/main.py --real-vla smolvla
python src/main.py --real-vla openvla
python src/main.py --real-vla octo

# Enable event camera
python src/main.py --event-camera

ROS2 Options

bash
# Enable ROS2 publishing
python src/main.py --ros2

# Enable Nav2 navigation
python src/main.py --nav2

SLAM & Navigation

bash
# Enable visual odometry
python src/main.py --visual-odom

# Convert depth to laser scan
python src/main.py --depth-to-scan

# Enable full SLAM
python src/main.py --slam
